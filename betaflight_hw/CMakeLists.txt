cmake_minimum_required(VERSION 3.16)
project(betaflight_hw_diagnostics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
set(AGILIB "${CMAKE_CURRENT_LIST_DIR}/../agilib")
add_library(hw_transport "${AGILIB}/src/bridge/betaflight/betaflight_msp_bridge.cpp")
target_include_directories(hw_transport PUBLIC "${AGILIB}/include")
target_compile_options(hw_transport PUBLIC -fno-finite-math-only -Wall -Wextra -Werror)
target_link_libraries(hw_transport PUBLIC Threads::Threads)
add_executable(agilicious_betaflight_hw "${AGILIB}/apps/betaflight_hw.cpp")
target_link_libraries(agilicious_betaflight_hw PRIVATE hw_transport)
add_executable(betaflight_hw_test "${AGILIB}/apps/tests/betaflight_hw_test.cpp")
target_link_libraries(betaflight_hw_test PRIVATE hw_transport util)
enable_testing()
add_test(NAME betaflight_hw_test COMMAND betaflight_hw_test)
