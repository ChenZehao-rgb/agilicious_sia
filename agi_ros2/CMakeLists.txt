cmake_minimum_required(VERSION 3.16)
project(agi_ros2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
foreach(dep diagnostic_msgs rclcpp sensor_msgs geometry_msgs nav_msgs std_msgs rosgraph_msgs builtin_interfaces)
  find_package(${dep} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/Rtk.msg" "msg/Authority.msg" "msg/Health.msg"
  "msg/Navigation.msg" "msg/LocalNavigation.msg" "msg/NavigationOrigin.msg"
  "msg/ComputationStatus.msg" "msg/MspState.msg"
  "msg/Heading.msg" "msg/FusedState.msg" "msg/SafetyEvidence.msg"
  "msg/ControlCommand.msg" "msg/OutputStatus.msg" "msg/MspEvent.msg"
  DEPENDENCIES std_msgs geometry_msgs builtin_interfaces)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_BENCH OFF CACHE BOOL "" FORCE)
set(BUILD_BETAFLIGHT_SITL OFF CACHE BOOL "" FORCE)
set(BUILD_BETAFLIGHT_HW OFF CACHE BOOL "" FORCE)
set(UNSAFE_MATH OFF CACHE BOOL "" FORCE)
set(AGILIB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../agilib")
add_subdirectory(${AGILIB_ROOT} ${CMAKE_CURRENT_BINARY_DIR}/agilib)
find_package(Eigen3 REQUIRED)
add_library(ros2_node_common STATIC src/node_common.cpp)
target_include_directories(ros2_node_common PUBLIC include ${AGILIB_ROOT}/include
  ${AGILIB_ROOT}/apps ${EIGEN3_INCLUDE_DIRS})
target_link_libraries(ros2_node_common agilib Eigen3::Eigen "${typesupport}")
ament_target_dependencies(ros2_node_common rclcpp builtin_interfaces)
add_executable(state_fusion_node src/state_fusion_main.cpp
  src/state_fusion_node.cpp ${AGILIB_ROOT}/apps/companion_ahrs.cpp)
add_executable(control_node src/control_main.cpp src/control_node.cpp)
add_executable(command_output_node src/command_output_main.cpp
  src/command_output_node.cpp src/msp_telemetry.cpp)
foreach(target ros2_node_common state_fusion_node control_node command_output_node)
  # Match agilib's Eigen alignment across the ABI boundary.
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64)$" AND NOT DEFINED CATKIN_DEVEL_PREFIX)
    target_compile_options(${target} PRIVATE -march=native)
  endif()
endforeach()
foreach(target state_fusion_node control_node command_output_node)
  target_link_libraries(${target} ros2_node_common agilib acados_mpc agilib
    Eigen3::Eigen "${typesupport}")
  ament_target_dependencies(${target} rclcpp sensor_msgs nav_msgs std_msgs)
  set_target_properties(${target} PROPERTIES
    INSTALL_RPATH "$ORIGIN/..;${ACADOS_ROOT}/lib"
    LINK_FLAGS "-Wl,--disable-new-dtags")
  install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
endforeach()
set_target_properties(agilib acados_mpc PROPERTIES INSTALL_RPATH "$ORIGIN;${ACADOS_ROOT}/lib" LINK_FLAGS "-Wl,--disable-new-dtags")
install(TARGETS agilib acados_mpc LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
add_executable(betaflight_msp_node src/msp_node.cpp src/msp_telemetry.cpp
  ${AGILIB_ROOT}/src/bridge/betaflight/betaflight_msp_bridge.cpp)
target_include_directories(betaflight_msp_node PRIVATE include ${AGILIB_ROOT}/include)
target_link_libraries(betaflight_msp_node "${typesupport}")
ament_target_dependencies(betaflight_msp_node rclcpp std_msgs)
target_compile_options(betaflight_msp_node PRIVATE -fno-finite-math-only -Wall -Wextra -Werror)
install(TARGETS betaflight_msp_node DESTINATION lib/${PROJECT_NAME})
set(AGI_ROS2_GAZEBO_DEFAULT ON)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|arm.*)$")
  set(AGI_ROS2_GAZEBO_DEFAULT OFF)
endif()
option(AGI_ROS2_GAZEBO "Build Gazebo Harmonic sensor adapter" ${AGI_ROS2_GAZEBO_DEFAULT})
if(AGI_ROS2_GAZEBO)
 find_package(gz-transport13 REQUIRED)
 find_package(gz-msgs10 REQUIRED)
 add_executable(gazebo_sensors src/gazebo_sensors.cpp)
 target_link_libraries(gazebo_sensors gz-transport13::core gz-msgs10::core "${typesupport}")
 ament_target_dependencies(gazebo_sensors rclcpp sensor_msgs nav_msgs rosgraph_msgs)
 install(TARGETS gazebo_sensors DESTINATION lib/${PROJECT_NAME})
endif()
add_executable(mavlink_sensor_node src/mavlink_sensor_node.cpp)
target_include_directories(mavlink_sensor_node SYSTEM PRIVATE third_party/mavlink)
target_link_libraries(mavlink_sensor_node "${typesupport}")
ament_target_dependencies(mavlink_sensor_node rclcpp sensor_msgs geometry_msgs diagnostic_msgs)
target_compile_options(mavlink_sensor_node PRIVATE -Wall -Wextra -Werror -fno-finite-math-only)
install(TARGETS mavlink_sensor_node DESTINATION lib/${PROJECT_NAME})
install(FILES README.md MAVLINK_SENSORS.md SHADOW_EVALUATION.md
  RUNTIME_WALKTHROUGH.zh-CN.md HARDWARE_ADAPTATION_VALIDATION.zh-CN.md RATE_MPC_VALIDATION.zh-CN.md
  DESTINATION share/${PROJECT_NAME})
install(DIRECTORY test/results/rate_mpc_20260922 DESTINATION share/${PROJECT_NAME}/test/results)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(DIRECTORY ${AGILIB_ROOT}/params/ DESTINATION share/${PROJECT_NAME}/params)
install(PROGRAMS scripts/sim_rc.py scripts/gnss_adapter.py scripts/msp_evidence.py scripts/summarize_shadow_bag.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/shadow_support.py DESTINATION lib/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
